cmake_minimum_required(VERSION 3.31)
project(IVTotalVerification LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 REQUIRED COMPONENTS Core Gui Widgets Test Network)
get_filename_component(SRC "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
set(POKEFINDER_BUILD_DIR "" CACHE PATH "Existing compatible PokeFinder+ build directory")
if(NOT POKEFINDER_BUILD_DIR)
    message(FATAL_ERROR "Set POKEFINDER_BUILD_DIR to the existing PokeFinder+ build")
endif()
set(BLD "${POKEFINDER_BUILD_DIR}")
add_executable(host_regression host_regression.cpp)
target_include_directories(host_regression PRIVATE ${SRC})
target_link_libraries(host_regression PRIVATE ${BLD}/PokeFinderPlus/PokeFinderPlusPluginManager.lib Qt6::Widgets)
add_executable(proxy_tests proxy_tests.cpp ${SRC}/PokeFinderPlus/IVTotal/IVTotalProxyModel.cpp ${SRC}/PokeFinderPlus/IVTotal/IVTotalProxyModel.hpp)
target_include_directories(proxy_tests PRIVATE ${SRC})
target_link_libraries(proxy_tests PRIVATE Qt6::Widgets Qt6::Test)
add_executable(integration integration.cpp)
target_include_directories(integration PRIVATE ${SRC})
target_link_libraries(integration PRIVATE ${BLD}/PokeFinderPlus/PokeFinderPlusPluginManager.lib ${BLD}/Form/PokeFinderForm.lib ${BLD}/Model/PokeFinderModel.lib ${BLD}/Core/PokeFinderCore.lib ${BLD}/Core/External/zstd/lib/zstd_static.lib Qt6::Widgets Qt6::Network Qt6::Test)
